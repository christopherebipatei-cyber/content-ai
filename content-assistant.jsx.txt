// ─── API ──────────────────────────────────────────────────────────────────────
async function callClaude(messages, system = SYSTEM_PROMPT) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ model: "claude-sonnet-4-20250514", max_tokens: 1000, system, messages }),
  });
  const data = await res.json();
  return data.content?.map((b) => b.text || "").join("") || "";
}

// ─── Post Unit (tabbed output) ────────────────────────────────────────────────
function PostUnit({ data, onNew }) {
  const [tab, setTab] = useState("card");
  const [downloading, setDownloading] = useState(false);
  const [copiedCap, setCopiedCap] = useState(false);
  const [copiedHash, setCopiedHash] = useState(false);

  const copy = (text, setter) => { navigator.clipboard.writeText(text); setter(true); setTimeout(() => setter(false), 2000); };

  const download = async () => {
    setDownloading(true);
    const card = document.getElementById("quote-card");
    const w = card.offsetWidth, h = card.offsetHeight;
    const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="${w}" height="${h}"><foreignObject width="100%" height="100%"><div xmlns="http://www.w3.org/1999/xhtml" style="width:${w}px;height:${h}px;background:#0d0d0d;overflow:hidden;"><div style="height:5px;background:linear-gradient(90deg,#ef4444,#f59e0b,#fde047)"></div><div style="padding:24px;font-family:serif;"><div style="font-size:9px;color:#f59e0b;letter-spacing:2.5px;text-transform:uppercase;font-family:sans-serif;font-weight:700;margin-bottom:16px;">${data.tag}</div><p style="font-size:20px;color:#fff;font-weight:900;line-height:1.35;margin:0 0 16px;">${data.quote}</p><div style="border-left:2px solid #2a2a2a;padding-left:12px;"><p style="font-size:11px;color:#555;font-family:sans-serif;font-style:italic;margin:0;">${data.subtext}</p></div></div></div></foreignObject></svg>`;
    const blob = new Blob([svg], { type: "image/svg+xml" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a"); a.href = url; a.download = "quote-card.svg"; a.click();
    URL.revokeObjectURL(url);
    setDownloading(false);
  };

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 14, animation: "fadeUp 0.3s ease" }}>
      <div style={{ display: "flex", gap: 2, background: "#111", borderRadius: 9, padding: 3, border: "1px solid #1a1a1a" }}>
        {[{ id: "card", label: "Quote Card" }, { id: "caption", label: "Caption" }, { id: "hashtags", label: "Hashtags" }].map((t) => (
          <button key={t.id} onClick={() => setTab(t.id)} style={{ flex: 1, background: tab === t.id ? "#1e1e1e" : "transparent", border: "none", borderRadius: 6, padding: "7px 0", color: tab === t.id ? "#f59e0b" : "#3a3a3a", fontSize: 11, cursor: "pointer", fontFamily: "'Lato',sans-serif", fontWeight: tab === t.id ? 700 : 400, transition: "all 0.2s" }}>
            {t.label}
          </button>
        ))}
      </div>

      {tab === "card" && <QuoteCard quote={data.quote} subtext={data.subtext} tag={data.tag} onDownload={download} downloading={downloading} />}

      {tab === "caption" && (
        <div style={{ background: "#111", border: "1px solid #1e1e1e", borderRadius: 14, padding: 18 }}>
          <p style={L}>ready-to-post caption</p>
          <p style={{ color: "#e5e5e5", fontSize: 14, lineHeight: 1.85, fontFamily: "'Lato',sans-serif", whiteSpace: "pre-wrap", margin: "0 0 14px" }}>{data.caption}</p>
          <button onClick={() => copy(data.caption, setCopiedCap)} style={copyBtn(copiedCap)}>{copiedCap ? "✓ copied" : "copy caption"}</button>
        </div>
      )}

      {tab === "hashtags" && (
        <div style={{ background: "#111", border: "1px solid #1e1e1e", borderRadius: 14, padding: 18 }}>
          <p style={L}>hashtag set</p>
          <div style={{ display: "flex", flexWrap: "wrap", gap: 7, marginBottom: 14 }}>
            {data.hashtags.split(" ").filter(Boolean).map((h, i) => (
              <span key={i} style={{ background: "#1a1a1a", border: "1px solid #2a2a2a", borderRadius: 6, padding: "4px 10px", color: "#f59e0b", fontSize: 11, fontFamily: "'Lato',sans-serif" }}>{h}</span>
            ))}
          </div>
          <button onClick={() => copy(data.hashtags, setCopiedHash)} style={copyBtn(copiedHash)}>{copiedHash ? "✓ copied" : "copy all hashtags"}</button>
        </div>
      )}

      <button onClick={onNew} style={{ background: "transparent", border: "1px solid #1e1e1e", borderRadius: 10, padding: "11px 0", color: "#555", fontSize: 12, cursor: "pointer", fontFamily: "'Lato',sans-serif", transition: "all 0.2s" }}>
        ↻ generate another
      </button>
    </div>
  );
}

// ─── Generate Mode ────────────────────────────────────────────────────────────
function GenerateMode({ onUse, usageBlocked }) {
  const [pillar, setPillar] = useState("mindset");
  const [platform, setPlatform] = useState("tiktok");
  const [topic, setTopic] = useState("");
  const [result, setResult] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  const generate = async () => {
    if (loading || usageBlocked) { if (usageBlocked) onUse(); return; }
    setLoading(true); setResult(null); setError(""); onUse();
    const p = PILLARS.find((x) => x.id === pillar);
    const pl = PLATFORMS.find((x) => x.id === platform);
    const prompt = `Create a ready-to-post content unit for the "${p.label}" pillar on ${pl.label}.${topic ? ` Angle/topic: ${topic}` : ""}\n\nReturn ONLY the JSON object. No markdown.`;
    try {
      const raw = await callClaude([{ role: "user", content: prompt }]);
      const parsed = JSON.parse(raw.replace(/```json|```/g, "").trim());
      if (!parsed.quote) throw new Error();
      setResult({ ...parsed, tag: parsed.tag || p.tag });
    } catch { setError("couldn't parse response — try again."); }
    setLoading(false);
  };

  if (result) return <PostUnit data={result} onNew={() => setResult(null)} />;

  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 18 }}>
      <div>
        <label style={L}>content pillar</label>
        <div style={{ display: "flex", gap: 7, flexWrap: "wrap" }}>
          {PILLARS.map((p) => <button key={p.id} onClick={() => setPillar(p.id)} style={chip(pillar === p.id, p.color)}>{p.icon} {p.label}</button>)}
        </div>
      </div>
      <div>
        <label style={L}>platform</label>
        <div style={{ display: "flex", gap: 7, flexWrap: "wrap" }}>
          {PLATFORMS.map((p) => <button key={p.id} onClick={() => setPlatform(p.id)} style={chip(platform === p.id, "#f59e0b")}>{p.label}</button>)}
        </div>
      </div>
      <div>
        <label style={L}>topic or angle (optional)</label>
        <input value={topic} onChange={(e) => setTopic(e.target.value)} onKeyDown={(e) => e.key === "Enter" && generate()}
          placeholder="e.g. why broke people stay broke..."
          style={{ ...inputStyle, width: "100%", boxSizing: "border-box" }} />
      </div>
      {error && <p style={{ fontSize: 12, color: "#ef4444", fontFamily: "'Lato',sans-serif" }}>{error}</p>}
      <button onClick={generate} disabled={loading} style={genBtn(loading, usageBlocked)}>
        {usageBlocked ? "🔒 upgrade to generate" : loading ? "building your post..." : "✦ create ready-to-post content"}
      </button>
      {loading && <p style={{ textAlign: "center", fontSize: 11, color: "#2a2a2a", fontFamily: "'Lato',sans-serif" }}>generating quote · caption · hashtags</p>}
    </div>
  );
}

// ─── Chat Mode ────────────────────────────────────────────────────────────────
function ChatMode({ onUse, usageBlocked }) {
  const [messages, setMessages] = useState([{ role: "assistant", content: "what do you want to create today?" }]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const bottomRef = useRef(null);
  const chatSystem = `You are a content assistant for a faceless social media brand. Pillars: Money Psychology, AI & Productivity, Identity & Mindset. Voice: minimalist, lowercase, punchy. Help with content ideas, captions, hooks, strategy.`;

  useEffect(() => { bottomRef.current?.scrollIntoView({ behavior: "smooth" }); }, [messages, loading]);

  const send = async () => {
    if (!input.trim() || loading) return;
    if (usageBlocked) { onUse(); return; }
    const userMsg = { role: "user", content: input.trim() };
    const next = [...messages, userMsg];
    setMessages(next); setInput(""); setLoading(true); onUse();
    try {
      const reply = await callClaude(next.map((m) => ({ role: m.role, content: m.content })), chatSystem);
      setMessages([...next, { role: "assistant", content: reply }]);
    } catch { setMessages([...next, { role: "assistant", content: "something went wrong. try again." }]); }
    setLoading(false);
  };

  return (
    <div style={{ display: "flex", flexDirection: "column", height: "100%" }}>
      <div style={{ flex: 1, overflowY: "auto", padding: "12px 0", display: "flex", flexDirection: "column", gap: 12 }}>
        {messages.map((m, i) => (
          <div key={i} style={{ display: "flex", justifyContent: m.role === "user" ? "flex-end" : "flex-start" }}>

  );

// ─── Main App ─────────────────────────────────────────────────────────────────
export default function App() {
  const [user, setUser] = useState(null); // { email, isPro }
  const [mode, setMode] = useState("generate");
  const [usage, setUsage] = useState(0);
  const [showPaywall, setShowPaywall] = useState(false);

  const isPro = user?.isPro || false;
  const usageBlocked = !isPro && usage >= FREE_LIMIT;
  const remaining = Math.max(FREE_LIMIT - usage, 0);
  const barPct = Math.min((usage / FREE_LIMIT) * 100, 100);
  const barColor = usage >= FREE_LIMIT ? "#ef4444" : usage >= FREE_LIMIT - 1 ? "#f59e0b" : "#22c55e";

  const handleUse = () => {
    if (isPro) return;
    if (usage >= FREE_LIMIT) { setShowPaywall(true); return; }
    setUsage((u) => { const next = u + 1; if (next >= FREE_LIMIT) setTimeout(() => setShowPaywall(true), 800); return next; });
  };

  const handleLogout = async () => {
    await fakeSupabase.auth.signOut();
    setUser(null); setUsage(0); setMode("generate");
  };

  // ── Not logged in → show auth screen
  if (!user) return (
    <>
      <style>{`@import url('https://fonts.googleapis.com/css2?family=Playfair+Display:wght@700;900&family=Lato:wght@300;400;700&display=swap');*{box-sizing:border-box;margin:0;padding:0}body{background:#0d0d0d}@keyframes fadeUp{from{opacity:0;transform:translateY(10px)}to{opacity:1;transform:translateY(0)}}`}</style>
      <AuthScreen onLogin={setUser} />
    </>
  );

  // ── Logged in → show app
  return (
    <>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Playfair+Display:wght@700;900&family=Lato:wght@300;400;700&display=swap');
        *{box-sizing:border-box;margin:0;padding:0}body{background:#0d0d0d}
        ::-webkit-scrollbar{width:4px}::-webkit-scrollbar-thumb{background:#222;border-radius:4px}
        @keyframes blink{0%,100%{opacity:.3;transform:scale(.8)}50%{opacity:1;transform:scale(1.2)}}
        @keyframes fadeUp{from{opacity:0;transform:translateY(10px)}to{opacity:1;transform:translateY(0)}}
        .app{animation:fadeUp 0.3s ease}
      `}</style>

      {showPaywall && <PaywallModal onClose={() => setShowPaywall(false)} onUpgrade={() => { setUser({ ...user, isPro: true }); setShowPaywall(false); }} />}

      <div className="app" style={{ minHeight: "100vh", background: "#0d0d0d", display: "flex", justifyContent: "center", padding: "22px 16px" }}>
        <div style={{ width: "100%", maxWidth: 520 }}>

          {/* Header */}
          <div style={{ marginBottom: 18 }}>
            <div style={{ height: 3, background: "linear-gradient(90deg,#ef4444,#f59e0b,#fde047)", borderRadius: 2, marginBottom: 14 }} />
            <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
              <div>
                <h1 style={{ fontFamily: "'Playfair Display',serif", fontSize: 20, color: "#fff", fontWeight: 900 }}>content<span style={{ color: "#f59e0b" }}>.</span>ai</h1>
                <p style={{ fontSize: 10, color: "#2a2a2a", fontFamily: "'Lato',sans-serif", marginTop: 2 }}>
                  {user.email}
                  {isPro && <span style={{ marginLeft: 6, color: "#f59e0b" }}>· PRO</span>}
                </p>
              </div>
              <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
                {!isPro && (
                  <div style={{ display: "flex", alignItems: "center", gap: 7 }}>
                    <div style={{ width: 46, height: 3, background: "#1a1a1a", borderRadius: 2, overflow: "hidden" }}>
                      <div style={{ width: `${barPct}%`, height: "100%", background: barColor, borderRadius: 2, transition: "width 0.4s" }} />

                    <button onClick={() => setShowPaywall(true)} style={{ background: "#f59e0b22", border: "1px solid #f59e0b33", borderRadius: 6, padding: "4px 9px", color: "#f59e0b", fontSize: 10, cursor: "pointer", fontFamily: "'Lato',sans-serif", fontWeight: 700 }}>PRO ↗</button>
                  </div>
                )}
                <button onClick={handleLogout} style={{ background: "transparent", border: "1px solid #1a1a1a", borderRadius: 6, padding: "5px 10px", color: "#333", fontSize: 10, cursor: "pointer", fontFamily: "'Lato',sans-serif" }}>sign out</button>
              </div>
            </div>
          </div>

          {/* Tabs */}
          <div style={{ display: "flex", gap: 2, background: "#111", borderRadius: 10, padding: 3, border: "1px solid #1a1a1a", marginBottom: 12 }}>
            {MODES.map((m) => (

            ))}

            {mode === "ideas" && <IdeasMode onUse={handleUse} usageBlocked={usageBlocked} />}

              <div>

              </div>
              <span style={{ color: "#f59e0b" }}>→</span>
            </div>
          )}

          <p style={{ textAlign: "center", marginTop: 14, fontSize: 10, color: "#161616", fontFamily: "'Lato',sans-serif" }}>
            money psychology · ai & productivity · identity & mindset
          </p>
        </div>
      </div>
    </>
  );
}
